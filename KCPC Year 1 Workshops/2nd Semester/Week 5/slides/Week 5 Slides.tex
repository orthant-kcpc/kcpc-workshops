\section{Dijkstra's Algorithm}
% ============================================================

\begin{frame}{Relaxation: Improve a Best-Known Route}
  \begin{itemize}
    \item $\mathrm{dist}[v]$ is the cheapest route \textbf{found so far} to $v$.
    \item Start with $\mathrm{dist}[s]=0$; every other label is $\infty$.
    \item Relax an edge $u\to v$ of weight $w$:
  \end{itemize}
  \[
    \mathrm{candidate}=\mathrm{dist}[u]+w.
  \]
  If $\mathrm{candidate}<\mathrm{dist}[v]$, update $\mathrm{dist}[v]$ and push $(\mathrm{dist}[v],v)$.
  \medskip
  Use a \textbf{min-heap}: it removes the smallest distance, not the oldest entry contrary to BFS.
\end{frame}

\begin{frame}{The Invariant That Makes Dijkstra Safe}
  \textbf{Invariant:} when the current minimum entry for $u$ is popped, $\mathrm{dist}[u]$ is final.
  \begin{itemize}
    \item Any cheaper route to $u$ must leave the popped set through some unpopped vertex $x$.
    \item With nonnegative weights, its cost up to $x$ is at most its total cost, so $\mathrm{dist}[x]<\mathrm{dist}[u]$.
    \item Then $x$ would have been popped before $u$: a contradiction.
  \end{itemize}
  \medskip
  A negative edge breaks this argument. Do \textbf{not} use Dijkstra on negative-weight graphs.
\end{frame}

\begin{frame}[fragile]{Dijkstra in Code (C++ and Python)}
  \begin{columns}[t]
    \begin{column}{0.49\textwidth}
      \textbf{C++ (\texttt{priority\_queue})}
\begin{lstlisting}[language=C++]
vector<ll> dist(n, INF);
priority_queue<PII, vector<PII>,
               greater<PII>> pq;
dist[src] = 0;
pq.push({0, src});
while (!pq.empty()) {
  auto [d, u] = pq.top();
  pq.pop();
  if (d != dist[u]) continue;
  for (auto [v, w] : adj[u]) {
    if (d + w < dist[v]) {
      dist[v] = d + w;
      pq.push({dist[v], v});
    }
  }
}
\end{lstlisting}
    \end{column}
    \begin{column}{0.49\textwidth}
      \textbf{Python (\texttt{heapq})}
\begin{lstlisting}[language=Python]
def dijkstra(adj, src):
  dist = [float("inf")] * len(adj)
  dist[src] = 0
  heap = [(0, src)]
  while heap:
    d, u = heappop(heap)
    if d != dist[u]:
      continue
    for v, w in adj[u]:
      nd = d + w
      if nd < dist[v]:
        dist[v] = nd
        heappush(heap, (nd, v))
  return dist
\end{lstlisting}
    \end{column}
  \end{columns}
  \medskip
  {\scriptsize $\mathrm{adj}[u]$ stores $(v,w)$ pairs. In C++, \texttt{ll} is \texttt{long long} and \texttt{PII} is \texttt{pair<ll,int>}.}
\end{frame}

\begin{frame}{Stale Entries and the Stopping Rule}
  \begin{itemize}
    \item Heap entries are \textbf{snapshots} $(\mathrm{distance}, \text{vertex})$. Improving a label pushes a new snapshot; the old one stays.
    \item A snapshot with $d\ne\mathrm{dist}[u]$ is \textbf{stale}: skip it without scanning neighbours.
    \item \textbf{One target?} Stop when the target is \emph{popped}, after the stale check.
    \item \textbf{All distances?} Run until the heap is empty.
    \item \textbf{Complexity?} $O(V+E\log E)$ time and $O(V+E)$ space.
  \end{itemize}
  \medskip
  \alert{Do not stop on first discovery or mark a vertex final on insertion.}
\end{frame}

\section{0--1 BFS}
% ============================================================

\begin{frame}{Binary Costs Do Not Need a Heap}
  Suppose every edge costs \textbf{0 or 1}.
  \begin{itemize}
    \item From a route of cost $d$, the next cost is only $d$ or $d+1$.
    \item A \textbf{deque} lets us insert at either end and pop from the front.
    \item A cost-0 improvement goes to the \textbf{front}: same cost layer.
    \item A cost-1 improvement goes to the \textbf{back}: next cost layer.
  \end{itemize}
  \medskip
  This is \textbf{0--1 BFS}: Dijkstra's ordering, without a heap.
  Ordinary FIFO BFS would count edges, not total cost.
\end{frame}

\begin{frame}{The Deque Invariant}
  \textbf{Store snapshots $(d,u)$}, just as in the heap.
  \[
    \underbrace{d,\ d,\ d}_{\text{front level}}
    \quad
    \underbrace{d+1,\ d+1}_{\text{next level}}
  \]
  \begin{itemize}
    \item Stored keys are nondecreasing, with at most \textbf{two adjacent levels}.
    \item On a current pop of cost $d$, front-inserting $d$ or back-inserting $d+1$ preserves this order.
    \item Skip a pop when $d\ne\mathrm{dist}[u]$, exactly as with Dijkstra.
  \end{itemize}
  \medskip
  Do not confuse \emph{stored keys} with the changing \texttt{dist} labels.
\end{frame}

\begin{frame}[fragile]{0--1 BFS in Code (C++ and Python)}
  \begin{columns}[t]
    \begin{column}{0.49\textwidth}
      \textbf{C++ (\texttt{deque})}
\begin{lstlisting}[language=C++]
vector<ll> dist(n, INF);
dist[src] = 0;
deque<PII> dq{{0, src}};
while (!dq.empty()) {
  auto [d, u] = dq.front();
  dq.pop_front();
  if (d != dist[u]) continue;
  for (auto [v, w] : adj[u]) {
    if (d + w < dist[v]) {
      dist[v] = d + w;
      if (w == 0)
        dq.push_front({dist[v], v});
      else
        dq.push_back({dist[v], v});
    }
  }
}
\end{lstlisting}
    \end{column}
    \begin{column}{0.49\textwidth}
      \textbf{Python (\texttt{collections.deque})}
\begin{lstlisting}[language=Python]
def zero_one_bfs(adj, src):
  dist = [float("inf")] * len(adj)
  dist[src] = 0
  dq = deque([(0, src)])
  while dq:
    d, u = dq.popleft()
    if d != dist[u]:
      continue
    for v, w in adj[u]:
      nd = d + w
      if nd < dist[v]:
        dist[v] = nd
        if w == 0:
          dq.appendleft((nd, v))
        else:
          dq.append((nd, v))
  return dist
\end{lstlisting}
    \end{column}
  \end{columns}
  \medskip
  {\scriptsize Strict $<$ queues only genuine improvements. $O(V+E)$ time and space.}
\end{frame}

\begin{frame}{Try It Yourself: Two Tempting Mistakes}
  Use the previous directed graph and the same edge-scan order.
  \begin{enumerate}
    \item Suppose we mark a vertex visited \emph{when enqueuing}, and
      never allow it to improve. What wrong distance reaches T?
    \item Keep the correct distance updates, but change the stale test
      to always process old entries. What work is repeated?
    \item Why must the relaxation test use $<$, not $\leq$, if a
      zero-cost cycle exists?
  \end{enumerate}
  \medskip
  Explain the role of \textbf{distance improvements}, not just queue order.
\end{frame}

\begin{frame}{Try It Yourself: Two Tempting Mistakes (Answer)}
  \begin{enumerate}
    \item With enqueue-time visited flags, S fixes A at 1 and B at 0.
      B cannot improve A, and discovers T at 1. A cannot improve T
      either. The wrong answer is \textbf{1}, instead of 0.
    \item Without the stale check, old $(1,A)$ rescans A's outgoing edge,
      and old $(1,T)$ processes T again. Strict relaxation prevents
      new improvements here, but these scans are unnecessary.
    \item With $\leq$, an edge on a zero-cost cycle can requeue a vertex
      at the \emph{same} distance forever. Strict $<$ queues only genuine
      improvements.
  \end{enumerate}
  \medskip
  The correct shortest route is $S\to B\to A\to T$, total cost 0.
  Discovery is not settlement: never make first discovery final.
\end{frame}

\begin{frame}{A Grid Is an Implicit Graph}
  \begin{itemize}
    \item \textbf{State}: a cell $(r,c)$; \textbf{edge}: one legal move.
    \item Generate neighbours when needed: usually
      $(r\pm1,c)$ and $(r,c\pm1)$.
    \item Reject out-of-bounds cells and any forbidden destinations.
    \item Define the \textbf{cost of the move}, then select the algorithm.
  \end{itemize}
  \medskip
  No need to build an adjacency list for every cell.
  An $R\times C$ four-neighbour grid has $RC$ vertices and $O(RC)$ edges.

  A rectangular drawing does \emph{not} imply unweighted BFS.
\end{frame}

\begin{frame}{Try It Yourself: What Are We Optimising?}
  A route's score can be extended by a sum, a product or a maximum. Compare the candidate routes:
  \begin{enumerate}
    \item Route A has step differences $2,9$; route B has $6,6$.
      Which is better under the \textbf{sum}? Which under the \textbf{maximum}?
    \item A direct route has success probability $0.75$.
      A two-edge route has probabilities $0.8$ and $0.9$.
      Which is better when we maximise the \textbf{product}?
  \end{enumerate}
  \medskip
  For each, state the operation used to extend a path score.
\end{frame}

\begin{frame}{Try It Yourself: What Are We Optimising? (Answer)}
  \begin{enumerate}
    \item \textbf{Sum:} $2+9=11<12=6+6$, so route A wins.
      \textbf{Maximum:} $\max(2,9)=9>6=\max(6,6)$, so route B wins.
      Extend by \emph{adding} for sums and by \emph{taking a maximum} for bottlenecks.
    \item \textbf{Product:} $0.8\cdot0.9=0.72<0.75$, so the direct route wins.
      Extend by \emph{multiplying}; maximise with a max-heap.
  \end{enumerate}
  \medskip
  The objective, not the graph, decides which route is best. In each case, a current best pop is final when extending a route cannot improve its score.
\end{frame}

\begin{frame}{Try It Yourself: Keep the Useful Arrival}
  At city $u$ we can arrive in two ways:
  \begin{itemize}
    \item cheaply but slowly: fee 3, time used 9;
    \item expensively but quickly: fee 6, time used 3.
  \end{itemize}
  The time limit is 10. The only move from $u$ to the target takes 3 more time units,
  and the target's entry fee is 2. The source fee and each entered city's fee are charged.
  \begin{enumerate}
    \item Which arrival at $u$ can finish? What is its total fee?
    \item Why does retaining only the cheapest fee at $u$ fail?
    \item How would you label the states and reject illegal transitions?
  \end{enumerate}
\end{frame}

\begin{frame}{Try It Yourself: Keep the Useful Arrival (Answer)}
  \begin{center}
  \small
  \begin{tabular}{@{}lccc@{}}
    \toprule
    Arrival at $u$ & Time used & Finish time & Result \\
    \midrule
    Cheap (fee 3) & 9 & $9+3=12$ & infeasible \\
    Fast (fee 6) & 3 & $3+3=6\le10$ & total fee $6+2=\boxed{8}$ \\
    \bottomrule
  \end{tabular}
  \end{center}
  \medskip
  \begin{itemize}
    \item Keeping only fee 3 at $u$ discards the arrival that can finish.
    \item Let $\mathrm{dist}[u][t]$ be the minimum fee to reach $u$ at time $t$.
      A move taking $\tau$ tries state $(v,t+\tau)$ with fee $\mathrm{dist}[u][t]+\mathrm{fee}[v]$.
  \end{itemize}
  Reject $t+\tau>10$. Initially $\mathrm{dist}[s][0]=\mathrm{fee}[s]$.
  Minimise over feasible arrival times at the target.
\end{frame}

\begin{frame}{Summary: Matching Models to Algorithms}
  \centering
  {\small
  \begin{tabular}{lll}
    \toprule
    \textbf{Need / edge costs} & \textbf{First choice} & \textbf{Time} \\
    \midrule
    Reachability only & DFS or BFS & $O(V+E)$ \\
    Unit / equal positive cost & BFS & $O(V+E)$ \\
    Costs 0 or 1 & 0--1 BFS & $O(V+E)$ \\
    General nonnegative costs & Dijkstra & heap logarithm \\
    General negative costs & e.g. Bellman--Ford & outside today's scope \\
    \bottomrule
  \end{tabular}}
  \medskip
  Multiple starts? Seed \textbf{all sources} at their correct initial scores.

  Time, fuel, keys or remaining moves affect future choices?
  They may belong in the \textbf{state}, not just in an edge weight.
\end{frame}
