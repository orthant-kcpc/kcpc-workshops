\section{Heuristic Methods}

\begin{frame}{What are Heuristics?}
    \pause
    A \textbf{heuristic} is a strategy or rule of thumb that helps
    you choose what to try when no solution is obvious. It does not
    guarantee success, but it can expose useful structure.

    The key question is: \textbf{what viewpoint makes the structure
    of the problem easier to see?}

    \pause
    \begin{itemize}
        \item \textbf{Wishful thinking:} imagine a simpler or nicer form.
        \item \textbf{Symmetry:} pair objects or equivalent cases.
        \item \textbf{Extreme principle:} inspect a boundary object.
        \item \textbf{Reduction:} transform the problem into a familiar one.
    \end{itemize}
\end{frame}

\section{Wishful Thinking}

\begin{frame}{Wishful Thinking: Simplify Until Solvable}
    \pause
    \begin{itemize}
        \item Assume the answer has a nice form. What would make this easy?
            \begin{itemize}
                \item ``If only the numbers were smaller...''
                \item ``If only the constraints were simpler...''
                \item ``What would the solution look like if I
                    already knew it?''
                \item Then find the reduction from the real problem to the
                    simplified one.
            \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Coordinator Example: Wish for a Square}
    Find the minimum of $x^2-6x+13$ for real $x$.

    \begin{center}
        \begin{tikzpicture}[x=0.7cm,y=0.23cm]
            \draw[->] (-0.3,0) -- (6.5,0) node[right] {$x$};
            \draw[->] (0,-0.3) -- (0,14) node[above] {$y$};
            \draw[kcpc-red,thick,domain=0:6,samples=40]
                plot (\x,{(\x-3)^2+4});
            \fill (3,4) circle (2pt);
            \node[above right,font=\small] at (3,4) {$(3,4)$};
            \node[below,font=\small] at (3,0) {3};
        \end{tikzpicture}
    \end{center}
    The drawing suggests a minimum. How can we \textbf{prove} it?
\end{frame}

\begin{frame}{Walkthrough: Complete the Square}
    \begin{itemize}
        \item \textbf{Wish:} a square would give an immediate lower bound.
        \item \textbf{Build it:} $(x-3)^2=x^2-6x+9$.
        \item \textbf{Account for the difference:}
            $x^2-6x+13=(x-3)^2+4$.
        \item \textbf{Prove the bound:} $(x-3)^2\ge0$, so the expression
            is at least 4.
        \item \textbf{Check attainability:} at $x=3$, its value is 4.
            Therefore the minimum is exactly 4.
    \end{itemize}
\end{frame}

\begin{frame}{Trio Checkpoint: Wishful Thinking}
    \textbf{Basic: Complete the Square}
    \[
        x^2+y^2\ge\frac{(x+y)^2}{2}\qquad(x,y\in\mathbb R).
    \]
    Rewrite the difference as a square. When is equality possible?

    \textbf{Easy, if ready:} Make the Sum Telescope (problem sheet).

    \textbf{Share:} a randomly selected trio explains its rewrite and
    why it proves the claim. We discuss each question attempted.
\end{frame}

\section{Symmetry}

\begin{frame}{Symmetry: Exploit Invariance Under Transformations}
    \pause
    \begin{itemize}
        \item Look for operations that leave the problem unchanged.
            \begin{itemize}
                \item Pairing strategy: if the opponent moves, reply
                    symmetrically.
                \item Canonical form: transform to a standard representative.
                \item Check that every paired response is legal.
            \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}[<1->]{Worked Example: Symmetry -- Pairing Game}
    \begin{columns}[c,onlytextwidth]
        \begin{column}{0.58\textwidth}
            Two players alternately place a \textbf{horizontal} domino
            on a $4\times4$ board. Dominoes cannot overlap.

            The player who cannot move loses. Show the second player
            has a winning strategy.

            What does reflection in the dashed line do to a move?
        \end{column}
        \begin{column}{0.38\textwidth}
            \centering\pairingboard{0}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}[<1->]{Pairing Game: Watch Two Rounds}
    \begin{columns}[c,onlytextwidth]
        \begin{column}{0.42\textwidth}
            \centering
            \only<1>{\pairingboard{1}}
            \only<2>{\pairingboard{2}}
            \only<3>{\pairingboard{3}}
            \only<4>{\pairingboard{4}}
        \end{column}
        \begin{column}{0.54\textwidth}
            \only<1>{\textbf{1.} A puts a domino in row 1.\par
                B's plan: use the same columns in row 4.}
            \only<2>{\textbf{2.} B reflects A's move.\par
                The occupied cells are now symmetric.}
            \only<3>{\textbf{3.} A next plays in row 2.\par
                Which cells should B use?}
            \only<4>{\textbf{4.} B plays in row 3.\par
                Symmetry is restored after the second round too.}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Solution: Pairing Game}
    \pause
    \begin{itemize}
        \item Mirror each move across the line between the middle two rows.
        \item This swaps rows 1 and 4, and rows 2 and 3.
        \item A horizontal domino stays in one row, so it never overlaps
            its reflection across this line.
        \item After each reply, the occupied squares are symmetric;
            the first player's next legal move has a legal, empty mirror.
        \item The second player never runs out of moves first, so wins.
    \end{itemize}
\end{frame}

\begin{frame}{Trio Checkpoint: Symmetry}
    \textbf{Basic: Pairing Game}

    Generalise our strategy to a $2r\times c$ board, with
    $r\ge1$, $c\ge2$, using horizontal dominoes only.

    \textbf{Explain:} which rows pair up? Why is the reply always empty
    and disjoint from the new domino?

    \textbf{Easy, if ready:} Coins on a Circular Table.
    What should happen at the centre?

    \textbf{Share:} randomly selected trios explain each attempted
    question; coordinators complete the legality argument.
\end{frame}

\section{Extreme Principle}

\begin{frame}{Extreme Principle: The Boundary Holds The Answer}
    \pause
    \begin{itemize}
        \item Consider the largest, smallest, leftmost, rightmost, or
            most extreme object.
            \begin{itemize}
                \item The extremal object has restricted neighbours.
                \item Often leads to contradiction or constructive
                    greedy choice.
                \item Works in geometry, combinatorics, number
                    theory, and algorithms.
            \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Coordinator Example: A Largest Element}
    Could a finite, non-empty set of \textbf{positive} integers be
    closed under addition? That means $a+b\in S$ for all $a,b\in S$,
    including $a=b$.

    \begin{center}
        \begin{tikzpicture}[x=0.65cm]
            \draw[->] (0,0) -- (10,0);
            \foreach \x in {1,2,3,4}
                \fill (\x,0) circle (2pt);
            \node[above] at (4,0.15) {$M=\max S$};
            \draw[kcpc-red,->,thick] (4,0.8) -- (8,0.8);
            \node[above,kcpc-red] at (6,0.8) {add $M$};
            \draw[kcpc-red,thick] (8,-0.12) -- (8,0.12);
            \node[below,kcpc-red] at (8,-0.12) {$2M>M$};
        \end{tikzpicture}
    \end{center}
\end{frame}

\begin{frame}{Solution: The Maximum Cannot Grow}
    \begin{itemize}
        \item Suppose such a set exists.
        \item Because it is finite and non-empty, it has a largest
            element $M$.
        \item Closure forces $M+M=2M$ to belong to the set.
        \item Positivity gives $2M>M$: a contradiction.
        \item No such set exists. The extreme element made the
            contradiction unavoidable.
    \end{itemize}
\end{frame}

\begin{frame}{Trio Checkpoint: Extreme Principle}
    \textbf{Basic: A Finite Set Closed Under Addition}

    Now allow \textbf{all integers}, including negative numbers and zero.
    If $S$ is finite, non-empty, and closed under addition,
    prove $S=\{0\}$.

    \textbf{Explain:} what does the maximum tell you? What extra
    information comes from the minimum?

    \textbf{Easy, if ready:} Minimum Degree Forces a Cycle.
    Use an endpoint of a longest path.

    \textbf{Share:} random trios explain each attempted question.
\end{frame}

\section{Reduction}

\begin{frame}{Reduction: Change The Representation}
    \pause
    \begin{itemize}
        \item Replace the problem with an equivalent one that is easier to see.
            \begin{itemize}
                \item Graph modelling: vertices and edges instead of story.
                \item Algebraic encoding: variables and equations
                    instead of logic.
                \item Modulo arithmetic: finite states instead of infinite.
                \item This is the bridge to competitive programming.
            \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}[<1->]{Graph Vocabulary for Reduction}
    \begin{columns}[c,onlytextwidth]
        \begin{column}{0.58\textwidth}
            \begin{itemize}
                \item A \textbf{vertex} represents an object or a complete
                    puzzle state.
                \item An \textbf{edge} represents a relationship or a legal move.
                \item A vertex's \textbf{degree} counts its incident edges.
                \item A \textbf{path} follows edges without repeating vertices;
                    a \textbf{cycle} closes back at its start.
            \end{itemize}
        \end{column}
        \begin{column}{0.38\textwidth}
            \centering
            \begin{tikzpicture}[scale=0.9,
                vertex/.style={circle,draw,fill=kcpc-bg,minimum size=0.6cm}]
                \node[vertex] (a) at (0,1.5) {A};
                \node[vertex] (b) at (2,1.5) {B};
                \node[vertex] (c) at (1,0) {C};
                \draw[thick] (a) -- (b) -- (c) -- (a);
                \node[below,font=\small] at (1,-0.5) {Each degree is 2.};
            \end{tikzpicture}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Worked Example: Reduction -- River Crossing}
    A farmer must take a wolf, a goat, and a cabbage across a river.
    The boat carries only the farmer and at most one item.

    The wolf cannot be left with the goat, and the goat cannot be left
    with the cabbage \textbf{without the farmer}. How can everyone cross safely?

    \begin{center}
        \riverbanks{Farmer, wolf,\\goat, cabbage}{Empty}{0}
    \end{center}
\end{frame}

\begin{frame}{River Crossing: Build the State Model}
    \begin{itemize}
        \item Encode a state as $(F,W,G,C)$: 0 means left, 1 means right.
            The boat is always with the farmer.
        \item A state is unsafe if $W=G\ne F$ or $G=C\ne F$.
        \item A legal crossing flips $F$ and, optionally, one item
            starting on the farmer's bank. The new state must be safe.
        \item The start is $0000$; the goal is $1111$.
        \item States are vertices; legal crossings are edges.
            We need a \textbf{path} between start and goal.
    \end{itemize}
\end{frame}

\begin{frame}{Full Walkthrough: River Crossing (1/2)}
    \begin{overprint}
        \onslide<1>
        \textbf{Start: $0000$.} Only the goat can go first: taking the
        wolf leaves goat/cabbage; taking cabbage leaves wolf/goat.
        \begin{center}\riverbanks{Farmer, wolf,\\goat, cabbage}{Empty}{0}\end{center}
        \onslide<2>
        \textbf{1. Goat over: $1010$.} The unattended bank has only
        wolf and cabbage, so it is safe.
        \begin{center}\riverbanks{Wolf, cabbage}{Farmer, goat}{1}\end{center}
        \onslide<3>
        \textbf{2. Farmer back alone: $0010$.} The goat can wait safely
        on its own. Bringing it back would undo our progress.
        \begin{center}\riverbanks{Farmer, wolf,\\cabbage}{Goat}{0}\end{center}
        \onslide<4>
        \textbf{3. Wolf over: $1110$.} Wolf and goat are safe together
        while the farmer is present. How can the farmer return?
        \begin{center}\riverbanks{Cabbage}{Farmer, wolf,\\goat}{1}\end{center}
    \end{overprint}
\end{frame}

\begin{frame}{Full Walkthrough: River Crossing (2/2)}
    \begin{overprint}
        \onslide<1>
        \textbf{4. Goat back: $0100$.} The farmer cannot leave wolf
        and goat alone. Bringing the goat back makes both banks safe.
        \begin{center}\riverbanks{Farmer, goat,\\cabbage}{Wolf}{0}\end{center}
        \onslide<2>
        \textbf{5. Cabbage over: $1101$.} Wolf and cabbage can be left
        together; the goat waits alone on the left.
        \begin{center}\riverbanks{Goat}{Farmer, wolf,\\cabbage}{1}\end{center}
        \onslide<3>
        \textbf{6. Farmer back alone: $0101$.} Neither unattended bank
        contains a dangerous pair.
        \begin{center}\riverbanks{Farmer, goat}{Wolf, cabbage}{0}\end{center}
        \onslide<4>
        \textbf{7. Goat over: $1111$.} Everyone is on the right bank.
        Every crossing was legal and every intermediate state safe.
        \begin{center}\riverbanks{Empty}{Farmer, wolf,\\goat, cabbage}{1}\end{center}
    \end{overprint}
\end{frame}

\begin{frame}{River Crossing: A Path Through Safe States}
    \begin{center}
        \begin{tikzpicture}[x=2.25cm,y=1.7cm,
            state/.style={draw,rounded corners,font=\small,
                minimum width=1.25cm,minimum height=0.6cm}]
            \node[state] (s0) at (0,1) {0000};
            \node[state] (s1) at (1,1) {1010};
            \node[state] (s2) at (2,1) {0010};
            \node[state] (s3) at (3,1) {1110};
            \node[state] (s4) at (3,0) {0100};
            \node[state] (s5) at (2,0) {1101};
            \node[state] (s6) at (1,0) {0101};
            \node[state] (s7) at (0,0) {1111};
            \foreach \a/\b/\label in {s0/s1/FG,s1/s2/F,s2/s3/FW,
                s4/s5/FC,s5/s6/F,s6/s7/FG}
                \draw[->,thick,kcpc-red] (\a) --
                    node[above,font=\scriptsize,kcpc-text] {\label} (\b);
            \draw[->,thick,kcpc-red] (s3) --
                node[right,font=\scriptsize,kcpc-text] {FG} (s4);
        \end{tikzpicture}
    \end{center}
    \small Edge labels show who crosses: F = farmer, W = wolf,
    G = goat, C = cabbage. This is \textbf{one solution path}, not
    the whole state graph.

    \textbf{Takeaway:} model the rules first, then search the legal states.
\end{frame}

\begin{frame}{Trio Checkpoint: Reduction}
    \textbf{Basic: An Acquaintance Graph}

    At a party of $n\ge2$ people, acquaintance is mutual. Prove
    that two people have the same number of acquaintances.

    \textbf{Try:} draw a small example, then list the possible degrees.
    Can degree 0 and degree $n-1$ occur together?

    \textbf{Easy, if ready:} The Bridges of K\"onigsberg.

    \textbf{Share:} random trios explain each attempted question;
    coordinators complete the counting argument if needed.
\end{frame}

\begin{frame}{Our Toolkit for Group Problems}
    \begin{itemize}[<1->]
        \item Wishful thinking simplifies the target structure.
        \item Symmetry pairs equivalent positions and cases.
        \item The extreme principle focuses on boundary objects.
        \item Reduction changes a problem into a familiar representation.
        \item Choose a viewpoint, test it on an example, then justify it.
    \end{itemize}
\end{frame}
