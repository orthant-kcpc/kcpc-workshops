\section{Heuristic Methods}

\begin{frame}{What are Heuristics?}
    \pause
    A \textbf{heuristic} is a strategy or rule of thumb that helps
    you choose what to try when no solution is obvious. It does not
    guarantee success, but it can expose useful structure.

    The key question is: \textbf{what viewpoint makes the structure
    of the problem easier to see?}

    \pause
    \begin{itemize}
        \item \textbf{Wishful thinking:} imagine a simpler or nicer form.
        \item \textbf{Symmetry:} pair objects or equivalent cases.
        \item \textbf{Extreme principle:} inspect a boundary object.
        \item \textbf{Reduction:} transform the problem into a familiar one.
    \end{itemize}
\end{frame}

\section{Wishful Thinking}

\begin{frame}{Wishful Thinking: Simplify Until Solvable}
    \pause
    \begin{itemize}
        \item Assume the answer has a nice form. What would make this easy?
            \begin{itemize}
                \item ``If only the numbers were smaller...''
                \item ``If only the constraints were simpler...''
                \item ``What would the solution look like if I
                    already knew it?''
                \item Then find the reduction from the real problem to the
                    simplified one.
            \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Worked Example: Wishful Thinking -- Sum of Squares}
    Construct infinitely many integer solutions to
    $x^2 + y^2 = 2z^2$ with $x,y,z > 0$.

    Wishful thinking: what if $x+y$ and $x-y$ had a nice structure?
\end{frame}

\begin{frame}{Solution: Sum Of Squares}
    \pause
    \begin{itemize}
        \item Set $x = u+v$, $y = u-v$.
        \item $x^2 + y^2 = (u+v)^2 + (u-v)^2 = 2u^2 + 2v^2 = 2z^2$.
        \item So $u^2 + v^2 = z^2$ (a Pythagorean triple!)
        \item Take $u = m^2-n^2$, $v = 2mn$, $z = m^2+n^2$.
        \item Then $x = (m^2-n^2) + 2mn$, $y = (m^2-n^2) - 2mn$.
        \item For example, fixing $n=1$ and taking $m\ge3$ gives
            infinitely many positive solutions.
    \end{itemize}
\end{frame}

\section{Symmetry}

\begin{frame}{Symmetry: Exploit Invariance Under Transformations}
    \pause
    \begin{itemize}
        \item Look for operations that leave the problem unchanged.
            \begin{itemize}
                \item Pairing strategy: if the opponent moves, reply
                    symmetrically.
                \item Canonical form: transform to a standard representative.
                \item Symmetric positions often have symmetric optimal moves.
            \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Worked Example: Symmetry -- Pairing Game}
    Two players alternately place a domino on a $4\times4$ board.
    Dominoes cannot overlap. The player who cannot move loses.
    Show the second player has a winning strategy.
\end{frame}

\begin{frame}{Solution: Pairing Game}
    \pause
    \begin{itemize}
        \item Mirror every move across the centre.
        \item The $4\times4$ board is centrally symmetric.
        \item If the first player places a domino covering $(x,y)$
            and $(x',y')$,
            the second player places one covering the centrally
            symmetric squares.
        \item This move is always legal because the board was symmetric and the
            first player's move was legal.
        \item The second player never runs out of moves first, so wins.
    \end{itemize}
\end{frame}

\section{Extreme Principle}

\begin{frame}{Extreme Principle: The Boundary Holds The Answer}
    \pause
    \begin{itemize}
        \item Consider the largest, smallest, leftmost, rightmost, or
            most extreme object.
            \begin{itemize}
                \item The extremal object has restricted neighbours.
                \item Often leads to contradiction or constructive
                    greedy choice.
                \item Works in geometry, combinatorics, number
                    theory, and algorithms.
            \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Worked Example: Extreme Principle -- A Finite Set}
    Let $S$ be a finite, non-empty set of integers such that
    $a+b\in S$ whenever $a,b\in S$.

    Prove that $S=\{0\}$.
\end{frame}

\begin{frame}{Solution: A Finite Set}
    \pause
    \begin{itemize}
        \item Examine the largest and smallest elements of $S$.
        \item Let $M$ be the largest element. If $M>0$, then
            $M+M=2M\in S$, contradicting maximality.
        \item Therefore $M\le 0$.
        \item Let $m$ be the smallest element. If $m<0$, then
            $m+m=2m\in S$, contradicting minimality.
        \item Therefore $m\ge 0$.
        \item Hence $m=M=0$, so $S=\{0\}$.
    \end{itemize}
\end{frame}

\section{Reduction}

\begin{frame}{Reduction: Change The Representation}
    \pause
    \begin{itemize}
        \item Replace the problem with an equivalent one that is easier to see.
            \begin{itemize}
                \item Graph modelling: vertices and edges instead of story.
                \item Algebraic encoding: variables and equations
                    instead of logic.
                \item Modulo arithmetic: finite states instead of infinite.
                \item This is the bridge to competitive programming.
            \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Worked Example: Reduction -- River Crossing}
    A farmer must take a wolf, a goat, and a cabbage across a river.
    The boat carries only the farmer and at most one item.

    The wolf cannot be left with the goat, and the goat cannot be left
    with the cabbage. How can everyone cross safely?
\end{frame}

\begin{frame}{Solution: River Crossing}
    \pause
    \begin{itemize}
        \item Represent a situation by which riverbank contains the farmer,
            wolf, goat, and cabbage.
        \item Keep only safe situations and connect two when one legal boat
            crossing changes one into the other.
        \item The puzzle is now a path problem from the all-left state to the
            all-right state.
        \item One path is: goat over, farmer back, wolf over, goat back,
            cabbage over, farmer back, goat over.
    \end{itemize}
\end{frame}
